cmake_minimum_required(VERSION 3.22)
project(FFmpegFreeUINative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
add_library(fui_core STATIC src/preset.cpp src/jobqueue.cpp src/platform.cpp src/resources.qrc)
target_include_directories(fui_core PUBLIC src)
target_link_libraries(fui_core PUBLIC Qt6::Core)
if(MSVC)
  target_compile_options(fui_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(fui_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(ffmpegfreeui src/main.cpp src/mainwindow.cpp)
target_link_libraries(ffmpegfreeui PRIVATE fui_core Qt6::Gui Qt6::Widgets)
if(WIN32)
  target_link_libraries(fui_core PRIVATE kernel32)
  set_target_properties(ffmpegfreeui PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS ffmpegfreeui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE.txt THIRD-PARTY-NOTICES.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/ffmpegfreeui-native)
install(FILES LICENSES/Upstream-MIT.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/ffmpegfreeui-native/LICENSES)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  install(PROGRAMS packaging/ffmpegfreeui-launch DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES packaging/ffmpegfreeui-native.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps RENAME ffmpegfreeui-native.png)
endif()
if(BUILD_TESTING)
  add_executable(fui_tests tests/core_tests.cpp)
  target_link_libraries(fui_tests PRIVATE fui_core)
  add_test(NAME native_core COMMAND fui_tests)
  set_tests_properties(native_core PROPERTIES TIMEOUT 120)
endif()
set(CPACK_PACKAGE_NAME ffmpegfreeui-native)
set(CPACK_PACKAGE_VENDOR "FFmpegFreeUI Native contributors")
set(CPACK_PACKAGE_CONTACT "FFmpegFreeUI Native contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native C++ FFmpeg frontend with Wayland and X11 support")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/Lake1059/FFmpegFreeUI")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.txt")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(CPACK_GENERATOR "DEB;RPM")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_DEBIAN_PACKAGE_RELEASE 1)
  set(CPACK_DEBIAN_PACKAGE_SECTION video)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  if(EXISTS "/etc/debian_version")
    find_program(DPKG_SHLIBDEPS_EXECUTABLE dpkg-shlibdeps REQUIRED)
  else()
    set(CPACK_GENERATOR RPM)
  endif()
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "ffmpeg, libqt6widgets6 (>= 6.4), qt6-wayland, qt6-qpa-plugins")
  set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "fonts-noto-cjk")
  set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
  set(CPACK_RPM_PACKAGE_RELEASE 1)
  set(CPACK_RPM_PACKAGE_LICENSE "AGPL-3.0-only AND MIT")
  set(CPACK_RPM_PACKAGE_GROUP "Applications/Multimedia")
  set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtbase-gui >= 6.4, qt6-qtwayland, /usr/bin/ffmpeg, /usr/bin/ffprobe")
  set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/icons /usr/share/icons/hicolor /usr/share/icons/hicolor/256x256 /usr/share/icons/hicolor/256x256/apps)
else()
  set(CPACK_GENERATOR ZIP)
endif()
include(CPack)
